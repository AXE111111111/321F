% =========================================================== 2  questions
\begin{frame}{Two questions, answered against two kinds of reference}
\bodytop
\begin{enumerate}\setlength{\itemsep}{0.8em}
\item For geometric Brownian motion, whose exact solution is known, do coupled Monte~Carlo experiments recover the
  theoretical strong and weak orders of Euler--Maruyama and Milstein once sampling uncertainty is quantified?
\item For a non-affine stochastic-volatility model without an exact transition law, does Euler--Maruyama refine at half order
  against nested numerical references, and can its weak payoff bias be resolved statistically?
\end{enumerate}
\vspace{0.95em}
{\leftskip=\leftmargini\lead{Why both.}\ Strong error is pathwise, weak error concerns expectations; their rates differ.\par}% content: report Sec. 1.1
\vspace{0.6em}
\hspace*{\leftmargini}\lead{Goal.}\ A trend counts only where it exceeds the Monte Carlo uncertainty.
\closing{In Direction 4, stability regions and adaptive steps are replaced by the exact--EM--Milstein comparison, coupled strong and
weak convergence, Monte Carlo confidence intervals and nested-grid validation; we add an analytic positivity analysis.}
\end{frame}

% =========================================================== 3  GBM schemes
\begin{frame}{GBM: the exact solution and two one-step schemes}
\bodytop\vspace{1.0mm}\vspace{-\abovedisplayskip}\vspace{-\baselineskip}
\begin{align*}
dS_t&=\mu S_t\,dt+\sigma S_t\,dW_t, & S_T&=S_0\exp\bigl[(\mu-\tfrac12\sigma^2)T+\sigma W_T\bigr],\\[0.2em]
S^{\mathrm{EM}}_{n+1}&=S_n\bigl(1+\mu h+\sigma\Delta W_n\bigr), &
S^{\mathrm{Mil}}_{n+1}&=S_n\bigl[1+\mu h+\sigma\Delta W_n+\tfrac12\sigma^2\bigl((\Delta W_n)^2-h\bigr)\bigr].
\end{align*}
Errors are measured on the same Brownian path,
\[
e_s(h)=\E\bigl|S^{(h)}_N-S_T\bigr|,\qquad e_w(h)=\bigl|\E S^{(h)}_N-\E S_T\bigr|,
\]
and theory predicts strong order $\tfrac12$ for Euler--Maruyama, $1$ for Milstein, and weak order $1$ for both.
\par\vspace{0.75em}% content: report Sec. 2.7 (positivity replaces stability)
\lead{Positivity.}\ The Euler--Maruyama factor $1+\mu h+\sigma\sqrt h\,Z$ can be negative for every $h$, the Milstein
factor only if $(\sigma^2-2\mu)h>1$; here $\sigma^2-2\mu=-0.06$ and $p_{\mathrm{EM}}(1/2)\approx2.12\times10^{-13}$.
\closing{$S_0=100$, $\mu=0.05$, $\sigma=0.20$, $T=1$; $M=2\times10^6$ paths, seed $32104$; $N=2,4,\dots,512$; coarse
increments are block sums of one $512$-step master path.}
\end{frame}

% =========================================================== 4  non-affine model
\begin{frame}{A non-affine volatility model with non-commutative noise}
\bodytop\vspace{1.8mm}\vspace{-\abovedisplayskip}\vspace{-\baselineskip}
\begin{gather*}
dX_t=\bigl(\mu-\tfrac12g(Y_t)^2\bigr)\,dt+g(Y_t)\,dW^{(1)}_t,\qquad
dY_t=\kappa(\theta-Y_t)\,dt+\xi\sqrt{1+Y_t^2}\,dW^{(2)}_t,\\[0.2em]
g(y)=\sigma_{\min}+\frac{\sigma_{\max}-\sigma_{\min}}{1+e^{-y}},\qquad d\langle W^{(1)},W^{(2)}\rangle_t=\rho\,dt,\qquad S_t=e^{X_t}.
\end{gather*}
\begin{proposition*}[non-commutativity]
With $q(y)=\xi\sqrt{1+y^2}$, the diffusion columns $b_1,b_2$ of $(X,Y)$ for independent drivers $B^{(1)}=W^{(1)}$, $B^{(2)}$ satisfy
\[
L^2b_1-L^1b_2=\bigl(\sqrt{1-\rho^2}\,q(Y)\,g'(Y),\,0\bigr)^{\!\top}\neq0.
\]
\end{proposition*}
Strong order one would in general need L\'evy areas (commutator $X$-component $0.0433$ at $Y=\theta$); we use
log-Euler--Maruyama, for which $S_{n+1}=e^{X_{n+1}}>0$.
\closing{Model: $\mu=0.05$, $\kappa=2$, $\theta=-0.2$, $\xi=0.6$, $\sigma_{\min}=0.10$, $\sigma_{\max}=0.50$.\\
Initial values, horizon and correlation: $S_0=100$, $Y_0=0$, $T=1$, $\rho=-0.7$; payoff $(S_T-100)^+$.}
\end{frame}

% =========================================================== 5  oracles
\begin{frame}{Two exact identities serve as oracles for the code}
\bodytop\vspace{-0.6mm}
\begin{proposition*}[GBM mean bias]
For Euler--Maruyama and Milstein, $\E S_N=S_0(1+\mu h)^N$, hence
\[
\delta(h)=S_0\bigl[(1+\mu h)^{T/h}-e^{\mu T}\bigr]=-\tfrac12S_0e^{\mu T}\mu^2T\,h+O(h^2).
\]
\end{proposition*}
For our parameters $\delta(1/2)=\analyticBiasHalf$; with $\mu=0$ the bias would vanish and $\varphi(s)=s$ could not reveal a
weak rate.% content: report Sec. 2.5
\par\vspace{0.85em}
\begin{proposition*}[log-EM preserves the mean]
For the non-affine log-Euler--Maruyama scheme and every $h=T/N$, \ $\E S_N=S_0e^{\mu T}=\E S_T$.
\end{proposition*}
\vspace{1.4em}
\noindent\lead{Remark.}\ The identity payoff therefore carries no discretisation bias in the non-affine model; its weak error
must be measured with a nonlinear payoff, here the call $(S_T-100)^+$.
\closing{Second identity as a code oracle: the paired difference $S^{(h)}_N-S^{(h_{\mathrm{ref}})}_N$ must be zero within error.}% content: report Remark 2.5, Sec. 4.6 (oracle passed)
\end{frame}

% =========================================================== 13  limitations
\begin{frame}{Limitations}
\bodytop
\begin{itemize}\setlength{\itemsep}{0.38em}
\item The non-affine errors are discrepancies against Euler--Maruyama references, not exact~errors; the reference gap
  is $\refRatio\%$ of the finest discrepancy.
\item The E2 weak slopes rest on coarse grids; the grid set was chosen after a first run.
\item The call payoff is not smooth and $e^X$ is not globally Lipschitz, so the non-affine rates are~empirical.
\item One parameter set; positivity was analysed only analytically; no Milstein scheme with L\'evy areas.
\item Slope errors are conditional jackknife SEs; timings come from one machine.
\end{itemize}
\closing{{\color{hairline}\rule{\textwidth}{0.4pt}}\par\vspace{0.6mm}\notesizesmall
AI use (report, Appendix A). Codex: numerical library and tests, review-driven revisions (resolved weak grids, jackknife SE);
on 10 Oct.\ (GPT-6 Astra, Ultra) the weak-grid refactor, independently re-checked. ChatGPT (GPT-6, Pro: review; GPT-6.1 Sol: language
polishing). Grok Bot: LaTeX typesetting, report restructuring and code packaging; drafted the supplementary code and two
derivations.}
\end{frame}
